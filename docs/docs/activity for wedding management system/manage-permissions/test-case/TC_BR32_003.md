### Method Logic

```csharp
private void PermissionState_UpdatePermission()
{
    foreach (var permission in PermissionStates)
    {
        bool wasAssigned = originalPermissions.Contains(permission.Key);
        bool isAssigned = permission.Value.IsChecked;

        if (wasAssigned && !isAssigned) // Remove
            RemovePermission(SelectedItem.GroupId, permission.Key);
        else if (!wasAssigned && isAssigned) // Add
            AddPermission(SelectedItem.GroupId, permission.Key);
    }
}
```
